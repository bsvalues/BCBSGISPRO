# 🟢 **2. Frontend React Code: “Live Only” Data Fetch**

**Replace every “dummy” or `const myData = [...]` block with real API fetches. Example for Parcels:**

```tsx
// client/src/components/RealParcelMap.tsx
import React, { useEffect, useState } from "react";

export function RealParcelMap() {
  const [parcels, setParcels] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch("/api/parcels")
      .then(res => res.json())
      .then(setParcels)
      .catch(() => setParcels([]))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <div className="p-4">Loading parcels...</div>;
  if (!parcels.length) return <div className="p-4 text-red-500">No parcels available.</div>;

  return (
    <div className="bg-zinc-800 rounded-lg p-4 shadow-xl">
      <div className="font-bold text-xl mb-2">Benton County Parcels</div>
      <ul>
        {parcels.map(parcel => (
          <li key={parcel.id} className="mb-1">
            {parcel.id} — {parcel.name || "Unnamed"} ({parcel.status})
          </li>
        ))}
      </ul>
    </div>
  );
}
```
